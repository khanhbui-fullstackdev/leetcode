	if total == target (9==17)❌
	else if total > target (17>9) ✅{
		rightIndex-- // 2
	}

	if total == target (13==9)❌
	else if total > target (13>9) ✅{
		rightIndex-- // 1
	}
}

(0<=1)
{
	total:= nums[leftIndex] + nums[rightIndex] 
	      = nums[0] + nums[1]  = 2 + 7 = 9
	
	if total == target (9==9)✅{
		indexes[0]=leftIndex+1
		indexes[1]=rightIndex+1
		break
	}
