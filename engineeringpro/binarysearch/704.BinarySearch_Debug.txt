leftIndex:=0
rightIndex:= 5

while (leftIndex<=rightIndex) (0<=5)✅{
    middleIndex:= (leftIndex + rightIndex) / 2 = (0 + 5) / 2 = 2
    if  (nums[2] == target) (3 == 9)❌
    else if (nums[2] < target) (3 < 9) ✅
    {
        leftIndex = middleIndex + 1 = 2 + 1 = 3
    }
}

while (leftIndex<=rightIndex) (3<=5)✅{
    middleIndex:= (leftIndex + rightIndex) / 2 = (3 + 5) / 2 = 4
    if  (nums[4] == target) (9 == 9)✅
    {
        return middleIndex//4
    }
}

KEY_NOTES
if (nums[middleIndex] < target) -> middle chắc chắn sai nên phải loại bỏ phạm vi tìm kiếm
{
    leftIndex = middleIndex + 1
}
else if (nums[middleIndex] > target)
{
    rightIndex = middleIndex - 1
}
