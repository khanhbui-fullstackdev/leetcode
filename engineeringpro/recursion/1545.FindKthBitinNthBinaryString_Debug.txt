 findKthBit(n int, k int)

// stack = [findKthBit(3,5)]
n = 3, k = 5{
	sLength(3) = 2^3 - 1 = 7
	midIndex = sLength/2 + 1 = 7/2 + 1 = 4
	if midIndex<k (4<5) ✅{
		reflectedPosition:= sLength - k + 1 = 7 - 5 + 1 = 3
		n = n - 1 = 3 - 1 = 2
		return 1 - findKthBit(n,reflectedPosition) = 1 - findKthBit(2,3)
	}
}

// stack = [findKthBit(3,5), findKthBit(2,3)]
n = 2, k = 3
{
	sLength(2) = 2^2 - 1 = 3
	midIndex = sLength/2 + 1 = 3/2 + 1 = 2
 	
	index   123  
	S2 = 	011
			       
	if midIndex<k (2<3) ✅{
		reflectedPosition:= sLength - k + 1 = 3 - 3 + 1 = 1
		n = n - 1 =  2 - 1 = 1
		return 1 - findKthBit(n,reflectedPosition) = 1 - findKthBit(1,1)
	}	
}

// stack = [findKthBit(3,5),findKthBit(2,3),findKthBit(1,1)]
n = 1, k = 1
{
	if n == 1 return 0 // pop stack = [findKthBit(3,5),findKthBit(2,3)]

// stack = [findKthBit(3,5), findKthBit(2,3)]
n = 2, k = 3
{
	sLength(2) = 2^2 - 1 = 3
	midIndex = sLength/2 + 1 = 3/2 + 1 = 2

		reflectedPosition:= sLength - k + 1 = 3 - 3 + 1 = 1
		n = n - 1 =  2 - 1 = 1
		return 1 - findKthBit(n,reflectedPosition) = 1 - findKthBit(1,1) = 1 - 0 = 1 // stack = [findKthBit(3,5)]
	}	
}

// stack = [findKthBit(3,5)]
